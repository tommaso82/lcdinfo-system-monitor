cmake_minimum_required(VERSION 3.20)
project(LCDInfoSystemMonitor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
endif()
enable_testing()
add_executable(render_test tests/render_test.cpp)
target_include_directories(render_test PRIVATE src)
add_test(NAME render_test COMMAND render_test)
if(WIN32)
  add_executable(LCDInfoSystemMonitor WIN32 src/main.cpp)
  target_compile_definitions(LCDInfoSystemMonitor PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(LCDInfoSystemMonitor PRIVATE winusb setupapi cfgmgr32 ole32 iphlpapi pdh shell32 user32 advapi32)
  if(MSVC)
    target_compile_options(LCDInfoSystemMonitor PRIVATE $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
    target_link_options(LCDInfoSystemMonitor PRIVATE $<$<CONFIG:Release>:/LTCG> /SUBSYSTEM:WINDOWS,6.01)
  endif()
endif()
