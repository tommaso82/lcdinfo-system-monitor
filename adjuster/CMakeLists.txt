cmake_minimum_required(VERSION 3.20)
project(LCDInfoDisplayAdjuster VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8)
endif()
enable_testing()
add_executable(settings_test settings_test.cpp)
add_test(NAME settings_test COMMAND settings_test)
if(WIN32)
  add_executable(LCDInfoDisplayAdjuster WIN32 main.cpp)
  target_compile_definitions(LCDInfoDisplayAdjuster PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_link_libraries(LCDInfoDisplayAdjuster PRIVATE winusb setupapi cfgmgr32 ole32 user32 gdi32 advapi32)
  if(MSVC)
    target_compile_options(LCDInfoDisplayAdjuster PRIVATE $<$<CONFIG:Release>:/O2> $<$<CONFIG:Release>:/GL>)
    target_link_options(LCDInfoDisplayAdjuster PRIVATE $<$<CONFIG:Release>:/LTCG> /SUBSYSTEM:WINDOWS,6.01)
  endif()
endif()
